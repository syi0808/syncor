# Syncor v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a cross-machine directory sync tool that monitors local directories, stores content via chkpt-core, and syncs via pluggable transport backends (Git first).

**Architecture:** Bottom-up build in 5 layers: Foundation (types, config, state DB) → Storage (save/restore pipelines, catalog merge) → Sync (conflict detection, transport, sync engine) → Automation (watcher, poller, daemon) → UI (CLI). Each layer builds on the previous and is independently testable.

**Tech Stack:** Rust, chkpt-core (storage engine), git2 (libgit2), tokio (async), notify (fsnotify), rusqlite (state DB), clap + dialoguer (CLI), serde + toml + serde_json (config/IPC), fs4 (flock), lz4_flex (compression)

```rust
// crates/syncor-core/src/link.rs
use serde::{Deserialize, Serialize};
use std::fmt;
use std::path::PathBuf;

/// Deterministic link identifier derived from repo + name.
#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct LinkId(String);

impl LinkId {
    pub fn from_parts(repo: &str, name: &str) -> Self {
        use std::hash::{Hash, Hasher};
        let mut hasher = std::collections::hash_map::DefaultHasher::new();
        repo.hash(&mut hasher);
        name.hash(&mut hasher);
        Self(format!("{:016x}", hasher.finish()))
    }

    pub fn as_str(&self) -> &str {
        &self.0
    }
}

impl fmt::Display for LinkId {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.write_str(&self.0)
    }
}

/// Whether this machine pushes (link) or pulls (connect).
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum LinkMode {
    Push,
    Pull,
}

/// Persisted link configuration.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct LinkInfo {
    pub id: LinkId,
    pub name: String,
    pub repo: String,
    pub local_dir: PathBuf,
    pub mode: LinkMode,
    pub poll_interval_secs: Option<u64>,
}

/// Runtime state of a link.
#[derive(Debug, Clone, Default, PartialEq, Eq)]
pub enum LinkState {
    #[default]
    Idle,
    Syncing,
    Conflict,
    Error(String),
}
```

```rust
// crates/syncor-core/src/config.rs
use crate::error::{Result, SyncorError};
use crate::link::LinkInfo;
use serde::{Deserialize, Serialize};
use std::path::{Path, PathBuf};

/// XDG-compliant paths for syncor data.
#[derive(Clone)]
pub struct SyncorPaths {
    config_base: PathBuf,
    data_base: PathBuf,
}

impl SyncorPaths {
    pub fn new() -> Self {
        let config_base = dirs::config_dir()
            .unwrap_or_else(|| PathBuf::from("~/.config"))
            .join("syncor");
        let data_base = dirs::data_dir()
            .unwrap_or_else(|| PathBuf::from("~/.local/share"))
            .join("syncor");
        Self { config_base, data_base }
    }

    pub fn with_home(home: &Path) -> Self {
        Self {
            config_base: home.join(".config").join("syncor"),
            data_base: home.join(".local").join("share").join("syncor"),
        }
    }

    pub fn config_dir(&self) -> &Path { &self.config_base }
    pub fn data_dir(&self) -> &Path { &self.data_base }
    pub fn config_file(&self) -> PathBuf { self.config_base.join("config.toml") }
    pub fn links_file(&self) -> PathBuf { self.config_base.join("links.toml") }
    pub fn socket_path(&self) -> PathBuf { self.data_base.join("syncor.sock") }
    pub fn pid_file(&self) -> PathBuf { self.data_base.join("daemon.pid") }
    pub fn log_file(&self) -> PathBuf { self.data_base.join("daemon.log") }

    pub fn link_dir(&self, link_id: &str) -> PathBuf {
        self.data_base.join(link_id)
    }

    pub fn link_repo_dir(&self, link_id: &str) -> PathBuf {
        self.link_dir(link_id).join("repo")
    }

    pub fn link_state_db(&self, link_id: &str) -> PathBuf {
        self.link_dir(link_id).join("state.db")
    }

    pub fn link_lock_file(&self, link_id: &str) -> PathBuf {
        self.link_dir(link_id).join("sync.lock")
    }

    pub fn ensure_dirs(&self) -> Result<()> {
        std::fs::create_dir_all(&self.config_base)?;
        std::fs::create_dir_all(&self.data_base)?;
        Ok(())
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SyncorConfig {
    #[serde(default = "default_debounce")]
    pub debounce_secs: u64,
    #[serde(default = "default_poll_interval")]
    pub default_poll_interval_secs: u64,
    #[serde(default)]
    pub auto_push: bool,
}

fn default_debounce() -> u64 { 2 }
fn default_poll_interval() -> u64 { 60 }

impl Default for SyncorConfig {
    fn default() -> Self {
        Self {
            debounce_secs: 2,
            default_poll_interval_secs: 60,
            auto_push: true,
        }
    }
}

impl SyncorConfig {
    pub fn load(path: &Path) -> Result<Self> {
        if !path.exists() {
            return Ok(Self::default());
        }
        let content = std::fs::read_to_string(path)?;
        toml::from_str(&content).map_err(|e| SyncorError::Config(e.to_string()))
    }

    pub fn save(&self, path: &Path) -> Result<()> {
        if let Some(parent) = path.parent() {
            std::fs::create_dir_all(parent)?;
        }
        let content = toml::to_string_pretty(self)
            .map_err(|e| SyncorError::Config(e.to_string()))?;
        std::fs::write(path, content)?;
        Ok(())
    }
}

/// Local registry of all links on this machine.
#[derive(Debug, Clone, Default, Serialize, Deserialize)]
pub struct LinksRegistry {
    #[serde(default)]
    pub links: Vec<LinkInfo>,
}

impl LinksRegistry {
    pub fn new() -> Self {
        Self { links: Vec::new() }
    }

    pub fn load(path: &Path) -> Result<Self> {
        if !path.exists() {
            return Ok(Self::new());
        }
        let content = std::fs::read_to_string(path)?;
        toml::from_str(&content).map_err(|e| SyncorError::Config(e.to_string()))
    }

    pub fn save(&self, path: &Path) -> Result<()> {
        if let Some(parent) = path.parent() {
            std::fs::create_dir_all(parent)?;
        }
        let content = toml::to_string_pretty(self)
            .map_err(|e| SyncorError::Config(e.to_string()))?;
        std::fs::write(path, content)?;
        Ok(())
    }

    pub fn add(&mut self, info: LinkInfo) -> Result<()> {
        // One dir, one link constraint
        if self.links.iter().any(|l| l.local_dir == info.local_dir) {
            return Err(SyncorError::LinkAlreadyExists(
                format!("directory {:?} is already linked", info.local_dir),
            ));
        }
        self.links.push(info);
        Ok(())
    }

    pub fn remove(&mut self, link_id: &str) -> Result<()> {
        let before = self.links.len();
        self.links.retain(|l| l.id.as_str() != link_id);
        if self.links.len() == before {
            return Err(SyncorError::LinkNotFound(link_id.to_string()));
        }
        Ok(())
    }

    pub fn get_by_name(&self, name: &str) -> Option<&LinkInfo> {
        self.links.iter().find(|l| l.name == name)
    }

    pub fn get_by_dir(&self, dir: &Path) -> Option<&LinkInfo> {
        self.links.iter().find(|l| l.local_dir == dir)
    }

    pub fn get_by_id(&self, id: &str) -> Option<&LinkInfo> {
        self.links.iter().find(|l| l.id.as_str() == id)
    }
}
```

```rust
// crates/syncor-core/src/sync/save.rs
use crate::error::Result;
use chkpt_core::index::{FileEntry, FileIndex};
use chkpt_core::scanner::{scan_workspace, ScannedFile};
use chkpt_core::store::blob::{bytes_to_hex, hash_content_bytes, read_or_mmap};
use chkpt_core::store::catalog::{
    BlobLocation, CatalogSnapshot, ManifestEntry, MetadataCatalog,
};
use chkpt_core::store::pack::PackWriter;
use chrono::Utc;
use lz4_flex::frame::FrameEncoder;
use std::io::Write;
use std::path::Path;
use uuid::Uuid;

#[derive(Debug)]
pub struct SaveResult {
    pub snapshot_id: String,
    pub files_scanned: usize,
    pub files_hashed: usize,
    pub bytes_compressed: u64,
}

pub struct SavePipeline;

impl SavePipeline {
    pub fn run(
        workspace: &Path,
        store_dir: &Path,
        message: Option<&str>,
    ) -> Result<SaveResult> {
        // Ensure store dirs exist
        let packs_dir = store_dir.join("packs");
        let trees_dir = store_dir.join("trees");
        let catalog_path = store_dir.join("catalog.sqlite");
        let index_path = store_dir.join("index.bin");
        std::fs::create_dir_all(&packs_dir)?;
        std::fs::create_dir_all(&trees_dir)?;

        // 1. Scan workspace
        let scanned = scan_workspace(workspace, None)?;
        let files_scanned = scanned.len();

        // 2. Load index for incremental detection
        let mut index = FileIndex::open(&index_path)?;

        // 3. Determine which files need hashing
        let mut to_hash: Vec<&ScannedFile> = Vec::new();
        for file in &scanned {
            if let Ok(Some(entry)) = index.get(&file.relative_path) {
                if entry.size == file.size
                    && entry.mtime_secs == file.mtime_secs
                    && entry.mtime_nanos == file.mtime_nanos
                {
                    continue; // unchanged
                }
            }
            to_hash.push(file);
        }
        let files_hashed = to_hash.len();

        // 4. Hash, compress, and pack changed files
        let mut pack_writer = PackWriter::new(&packs_dir)?;
        let mut blob_locations: Vec<([u8; 16], BlobLocation)> = Vec::new();
        let mut new_index_entries: Vec<FileEntry> = Vec::new();
        let mut bytes_compressed: u64 = 0;

        for file in &to_hash {
            let content = read_or_mmap(&file.absolute_path)?;
            let hash = hash_content_bytes(content.as_ref());

            // LZ4 compress
            let mut encoder = FrameEncoder::new(Vec::new());
            encoder.write_all(content.as_ref())?;
            let compressed = encoder.finish()?;
            bytes_compressed += compressed.len() as u64;

            let hash_hex = bytes_to_hex(&hash);
            pack_writer.add_pre_compressed(hash_hex, compressed)?;

            new_index_entries.push(FileEntry {
                path: file.relative_path.clone(),
                blob_hash: hash,
                size: file.size,
                mtime_secs: file.mtime_secs,
                mtime_nanos: file.mtime_nanos,
                inode: file.inode,
                mode: file.mode,
            });
        }

        // 5. Finish pack
        let pack_hash = if !pack_writer.is_empty() {
            Some(pack_writer.finish()?)
        } else {
            drop(pack_writer);
            None
        };

        // 6. Record blob locations
        if let Some(ref ph) = pack_hash {
            for entry in &new_index_entries {
                blob_locations.push((entry.blob_hash, BlobLocation {
                    pack_hash: Some(ph.clone()),
                    size: entry.size,
                }));
            }
        }

        // 7. Build manifest (all files — from index + new entries)
        let mut manifest: Vec<ManifestEntry> = Vec::new();
        // First, add unchanged files from index
        for file in &scanned {
            if let Ok(Some(entry)) = index.get(&file.relative_path) {
                // Check if this file was NOT re-hashed
                if !new_index_entries.iter().any(|e| e.path == file.relative_path) {
                    manifest.push(ManifestEntry {
                        path: entry.path.clone(),
                        blob_hash: entry.blob_hash,
                        size: entry.size,
                        mode: entry.mode,
                    });
                }
            }
        }
        // Then add newly hashed files
        for entry in &new_index_entries {
            manifest.push(ManifestEntry {
                path: entry.path.clone(),
                blob_hash: entry.blob_hash,
                size: entry.size,
                mode: entry.mode,
            });
        }
        manifest.sort_by(|a, b| a.path.cmp(&b.path));

        // 8. Open catalog and insert snapshot
        let catalog = MetadataCatalog::open(&catalog_path)?;
        let parent = catalog.latest_snapshot()?;

        let snapshot_id = Uuid::now_v7().to_string();
        let snapshot = CatalogSnapshot {
            id: snapshot_id.clone(),
            created_at: Utc::now(),
            message: message.map(|s| s.to_string()),
            parent_snapshot_id: parent.as_ref().map(|p| p.id.clone()),
            manifest_snapshot_id: None,
            root_tree_hash: None,
            stats: chkpt_core::store::snapshot::SnapshotStats {
            total_files: manifest.len() as u64,
            total_bytes: manifest.iter().map(|e| e.size).sum(),
            new_objects: files_hashed as u64,
        },
        };

        if !blob_locations.is_empty() {
            catalog.bulk_upsert_blob_locations(&blob_locations)?;
        }
        catalog.insert_snapshot(&snapshot, &manifest)?;

        // 9. Update index
        // Remove files that no longer exist
        let scanned_paths: std::collections::HashSet<&str> =
            scanned.iter().map(|f| f.relative_path.as_str()).collect();
        let all_indexed = index.all_paths()?;
        let removed: Vec<String> = all_indexed
            .into_iter()
            .filter(|p| !scanned_paths.contains(p.as_str()))
            .collect();
        index.apply_changes(&removed, &new_index_entries)?;

        Ok(SaveResult {
            snapshot_id,
            files_scanned,
            files_hashed,
            bytes_compressed,
        })
    }
}
```

```rust
// crates/syncor-core/src/sync/restore.rs
use crate::error::Result;
use chkpt_core::store::blob::bytes_to_hex;
use chkpt_core::store::catalog::MetadataCatalog;
use chkpt_core::store::pack::PackSet;
use std::path::Path;

#[derive(Debug)]
pub struct RestoreResult {
    pub files_restored: usize,
    pub files_removed: usize,
}

pub struct RestorePipeline;

impl RestorePipeline {
    pub fn run(
        snapshot_id: &str,
        store_dir: &Path,
        target_dir: &Path,
    ) -> Result<RestoreResult> {
        let catalog_path = store_dir.join("catalog.sqlite");
        let packs_dir = store_dir.join("packs");

        let catalog = MetadataCatalog::open(&catalog_path)?;
        let manifest = catalog.snapshot_manifest(snapshot_id)?;

        // Open all packs for reading
        let pack_set = PackSet::open_all(&packs_dir)?;

        // Build set of expected paths
        let expected_paths: std::collections::HashSet<String> =
            manifest.iter().map(|e| e.path.clone()).collect();

        // Restore each file from manifest
        let mut files_restored = 0;
        for entry in &manifest {
            let hash_hex = bytes_to_hex(&entry.blob_hash);
            let content = pack_set.read(&hash_hex)?;
            let file_path = target_dir.join(&entry.path);

            if let Some(parent) = file_path.parent() {
                std::fs::create_dir_all(parent)?;
            }
            std::fs::write(&file_path, content)?;

            // Restore file permissions on unix
            #[cfg(unix)]
            {
                use std::os::unix::fs::PermissionsExt;
                let perms = std::fs::Permissions::from_mode(entry.mode);
                std::fs::set_permissions(&file_path, perms)?;
            }

            files_restored += 1;
        }

        // Remove files not in manifest
        let mut files_removed = 0;
        for dir_entry in walkdir(target_dir)? {
            let relative = dir_entry
                .strip_prefix(target_dir)
                .unwrap()
                .to_string_lossy()
                .to_string();
            if !expected_paths.contains(&relative) {
                std::fs::remove_file(&dir_entry)?;
                files_removed += 1;
            }
        }

        // Clean up empty directories
        cleanup_empty_dirs(target_dir)?;

        Ok(RestoreResult {
            files_restored,
            files_removed,
        })
    }
}

/// Walk directory and return all file paths (not directories).
fn walkdir(dir: &Path) -> Result<Vec<std::path::PathBuf>> {
    let mut files = Vec::new();
    if !dir.exists() {
        return Ok(files);
    }
    for entry in std::fs::read_dir(dir)? {
        let entry = entry?;
        let path = entry.path();
        if path.is_dir() {
            files.extend(walkdir(&path)?);
        } else {
            files.push(path);
        }
    }
    Ok(files)
}

/// Remove empty directories bottom-up, but never the root.
fn cleanup_empty_dirs(root: &Path) -> Result<()> {
    cleanup_empty_dirs_inner(root, root)
}

fn cleanup_empty_dirs_inner(dir: &Path, root: &Path) -> Result<()> {
    if !dir.is_dir() {
        return Ok(());
    }
    for entry in std::fs::read_dir(dir)? {
        let entry = entry?;
        if entry.path().is_dir() {
            cleanup_empty_dirs_inner(&entry.path(), root)?;
        }
    }
    // Never remove the root directory itself
    if dir != root {
        let _ = std::fs::remove_dir(dir);
    }
    Ok(())
}
```

```rust
// crates/syncor-core/src/sync/catalog_merge.rs
use crate::error::Result;
use rusqlite::Connection;
use std::path::Path;

/// Merge remote catalog entries into local catalog using ATTACH + INSERT OR IGNORE.
/// Both paths must point to existing SQLite catalog databases.
pub fn merge_catalogs(local_path: &Path, remote_path: &Path) -> Result<()> {
    let conn = Connection::open(local_path)?;

    // Checkpoint WAL on local before merge
    conn.execute_batch("PRAGMA wal_checkpoint(TRUNCATE);")?;

    // Attach remote catalog
    conn.execute(
        "ATTACH DATABASE ?1 AS remote",
        [remote_path.to_str().unwrap()],
    )?;

    // Merge blob_index
    conn.execute_batch(
        "INSERT OR IGNORE INTO blob_index SELECT * FROM remote.blob_index;"
    )?;

    // Merge snapshots
    conn.execute_batch(
        "INSERT OR IGNORE INTO snapshots SELECT * FROM remote.snapshots;"
    )?;

    // Merge snapshot_files
    conn.execute_batch(
        "INSERT OR IGNORE INTO snapshot_files SELECT * FROM remote.snapshot_files;"
    )?;

    // Detach
    conn.execute_batch("DETACH DATABASE remote;")?;

    // Checkpoint WAL after merge
    conn.execute_batch("PRAGMA wal_checkpoint(TRUNCATE);")?;

    Ok(())
}

/// Checkpoint WAL to flush all data into the main db file.
/// Call this before copying catalog.sqlite for git commit.
pub fn checkpoint_wal(catalog_path: &Path) -> Result<()> {
    let conn = Connection::open(catalog_path)?;
    conn.execute_batch("PRAGMA wal_checkpoint(TRUNCATE);")?;
    Ok(())
}
```

```rust
// crates/syncor-core/src/sync/conflict.rs
use crate::error::Result;
use std::collections::{HashMap, HashSet};

pub type ManifestMap = HashMap<String, [u8; 16]>;

#[derive(Debug, Clone)]
pub enum FileAction {
    /// Apply remote version to local.
    ApplyRemote { path: String, remote_hash: [u8; 16] },
    /// Delete local file (remote deleted it).
    DeleteLocal { path: String },
    /// Conflict — needs resolution.
    Conflict(Conflict),
}

#[derive(Debug, Clone)]
pub struct Conflict {
    pub path: String,
    pub base_hash: Option<[u8; 16]>,
    pub local_hash: Option<[u8; 16]>,
    pub remote_hash: Option<[u8; 16]>,
}

#[derive(Debug, Clone)]
pub enum Resolution {
    KeepLocal,
    KeepRemote,
    Merged(Vec<u8>),
    Skip,
}

pub trait ConflictResolver {
    fn resolve(&self, conflict: &Conflict) -> Result<Resolution>;
}

/// Non-interactive resolver that always keeps local. Useful for testing.
pub struct KeepLocalResolver;

impl ConflictResolver for KeepLocalResolver {
    fn resolve(&self, _conflict: &Conflict) -> Result<Resolution> {
        Ok(Resolution::KeepLocal)
    }
}

/// Three-point comparison. Returns list of actions needed.
pub fn detect_conflicts(
    base: &ManifestMap,
    local: &ManifestMap,
    remote: &ManifestMap,
) -> Vec<FileAction> {
    let mut actions = Vec::new();
    let all_paths: HashSet<&String> = base.keys()
        .chain(local.keys())
        .chain(remote.keys())
        .collect();

    for path in all_paths {
        let b = base.get(path);
        let l = local.get(path);
        let r = remote.get(path);

        match (b, l, r) {
            // No change
            (Some(bh), Some(lh), Some(rh)) if bh == lh && lh == rh => {}
            // Both changed to same
            (Some(_), Some(lh), Some(rh)) if lh == rh => {}
            // Both deleted
            (Some(_), None, None) => {}
            // Remote changed, local unchanged
            (Some(bh), Some(lh), Some(rh)) if bh == lh && lh != rh => {
                actions.push(FileAction::ApplyRemote {
                    path: path.clone(),
                    remote_hash: *rh,
                });
            }
            // Local changed, remote unchanged
            (Some(bh), Some(lh), Some(rh)) if bh == rh && lh != rh => {
                // Keep local — nothing to do
            }
            // Both changed differently
            (Some(_bh), Some(lh), Some(rh)) => {
                actions.push(FileAction::Conflict(Conflict {
                    path: path.clone(),
                    base_hash: b.copied(),
                    local_hash: Some(*lh),
                    remote_hash: Some(*rh),
                }));
            }
            // Local added only
            (None, Some(_), None) => {} // keep local
            // Remote added only
            (None, None, Some(rh)) => {
                actions.push(FileAction::ApplyRemote {
                    path: path.clone(),
                    remote_hash: *rh,
                });
            }
            // Both added same
            (None, Some(lh), Some(rh)) if lh == rh => {}
            // Both added differently
            (None, Some(lh), Some(rh)) => {
                actions.push(FileAction::Conflict(Conflict {
                    path: path.clone(),
                    base_hash: None,
                    local_hash: Some(*lh),
                    remote_hash: Some(*rh),
                }));
            }
            // Local deleted, remote unchanged — keep local deletion (no action)
            (Some(bh), None, Some(rh)) if bh == rh => {}
            // Remote deleted, local unchanged
            (Some(bh), Some(lh), None) if bh == lh => {
                actions.push(FileAction::DeleteLocal { path: path.clone() });
            }
            // Local changed, remote deleted
            (Some(_), Some(lh), None) => {
                actions.push(FileAction::Conflict(Conflict {
                    path: path.clone(),
                    base_hash: b.copied(),
                    local_hash: Some(*lh),
                    remote_hash: None,
                }));
            }
            // Local deleted, remote changed
            (Some(_), None, Some(rh)) => {
                actions.push(FileAction::Conflict(Conflict {
                    path: path.clone(),
                    base_hash: b.copied(),
                    local_hash: None,
                    remote_hash: Some(*rh),
                }));
            }
            // Shouldn't happen
            (None, None, None) => {}
        }
    }

    actions
}
```

```rust
// crates/syncor-core/src/transport/git.rs
use crate::config::SyncorPaths;
use crate::error::{Result, SyncorError};
use crate::link::LinkInfo;
use crate::transport::{ConflictInfo, PullResult, PushResult, RemoteLinkInfo, SyncTransport};
use git2::{Repository, Signature};
use std::path::{Path, PathBuf};

pub struct GitTransport {
    paths: SyncorPaths,
}

impl GitTransport {
    pub fn new(paths: SyncorPaths) -> Self {
        Self { paths }
    }

    fn repo_dir(&self, link: &LinkInfo) -> PathBuf {
        self.paths.link_repo_dir(link.id.as_str())
    }

    fn open_repo(&self, link: &LinkInfo) -> Result<Repository> {
        Repository::open(self.repo_dir(link))
            .map_err(|e| SyncorError::Transport(format!("failed to open repo: {}", e)))
    }

    fn signature() -> Signature<'static> {
        Signature::now("syncor", "syncor@localhost").unwrap()
    }
}

impl SyncTransport for GitTransport {
    fn init_remote(&self, link: &LinkInfo) -> Result<()> {
        let repo_dir = self.repo_dir(link);
        if repo_dir.exists() {
            return Ok(()); // already cloned
        }
        std::fs::create_dir_all(&repo_dir)?;

        // Clone or init
        let _repo = match Repository::clone(&link.repo, &repo_dir) {
            Ok(repo) => repo,
            Err(_) => {
                // If clone fails (empty repo), init locally and add remote
                let repo = Repository::init(&repo_dir)
                    .map_err(|e| SyncorError::Transport(format!("git init failed: {}", e)))?;
                repo.remote("origin", &link.repo)
                    .map_err(|e| SyncorError::Transport(format!("add remote failed: {}", e)))?;
                repo
            }
        };

        // Create store directory structure
        let store_dir = repo_dir.join("stores").join(&link.name);
        std::fs::create_dir_all(store_dir.join("packs"))?;
        std::fs::create_dir_all(store_dir.join("trees"))?;

        // Create .gitignore to exclude machine-local files
        let gitignore_content = "# Machine-local files — do not sync\n\
                                  **/index.bin\n\
                                  **/*.sqlite-wal\n\
                                  **/*.sqlite-shm\n";
        std::fs::write(repo_dir.join(".gitignore"), gitignore_content)?;

        Ok(())
    }

    fn push(&self, link: &LinkInfo, _store_path: &Path) -> Result<PushResult> {
        let repo = self.open_repo(link)?;

        // Stage all changes
        let mut index = repo.index()
            .map_err(|e| SyncorError::Transport(format!("index error: {}", e)))?;
        index.add_all(["."], git2::IndexAddOption::DEFAULT, None)
            .map_err(|e| SyncorError::Transport(format!("add all error: {}", e)))?;
        index.write()
            .map_err(|e| SyncorError::Transport(format!("index write error: {}", e)))?;

        let tree_oid = index.write_tree()
            .map_err(|e| SyncorError::Transport(format!("write tree error: {}", e)))?;
        let tree = repo.find_tree(tree_oid)
            .map_err(|e| SyncorError::Transport(format!("find tree error: {}", e)))?;

        let sig = Self::signature();
        let parent = repo.head().ok().and_then(|h| h.peel_to_commit().ok());
        let parents: Vec<&git2::Commit> = parent.iter().collect();

        let oid = repo.commit(
            Some("HEAD"),
            &sig,
            &sig,
            "syncor: sync update",
            &tree,
            &parents,
        ).map_err(|e| SyncorError::Transport(format!("commit error: {}", e)))?;

        // Push to remote
        if let Ok(mut remote) = repo.find_remote("origin") {
            match remote.push(&["refs/heads/main:refs/heads/main"], None) {
                Ok(_) => {}
                Err(e) => {
                    // Check if non-fast-forward
                    if e.code() == git2::ErrorCode::NotFastForward {
                        return Ok(PushResult::Conflict {
                            details: ConflictInfo {
                                message: "non-fast-forward push rejected".to_string(),
                            },
                        });
                    }
                    return Err(SyncorError::Transport(format!("push error: {}", e)));
                }
            }
        }

        Ok(PushResult::Success {
            revision: oid.to_string(),
        })
    }

    fn pull(&self, link: &LinkInfo, _store_path: &Path) -> Result<PullResult> {
        let repo = self.open_repo(link)?;

        // Fetch
        let mut remote = repo.find_remote("origin")
            .map_err(|e| SyncorError::Transport(format!("find remote error: {}", e)))?;
        remote.fetch(&["main"], None, None)
            .map_err(|e| SyncorError::Transport(format!("fetch error: {}", e)))?;
        drop(remote);

        // Check if there are changes
        let fetch_head = match repo.find_reference("refs/remotes/origin/main") {
            Ok(r) => r,
            Err(_) => return Ok(PullResult::UpToDate),
        };
        let fetch_commit = fetch_head.peel_to_commit()
            .map_err(|e| SyncorError::Transport(format!("peel error: {}", e)))?;

        let local_head = repo.head().ok().and_then(|h| h.peel_to_commit().ok());

        if let Some(ref local) = local_head {
            if local.id() == fetch_commit.id() {
                return Ok(PullResult::UpToDate);
            }
        }

        // Fast-forward merge
        let refname = "refs/heads/main";
        repo.reference(refname, fetch_commit.id(), true, "syncor: pull")
            .map_err(|e| SyncorError::Transport(format!("ref update error: {}", e)))?;
        repo.set_head(refname)
            .map_err(|e| SyncorError::Transport(format!("set head error: {}", e)))?;
        repo.checkout_head(Some(git2::build::CheckoutBuilder::new().force()))
            .map_err(|e| SyncorError::Transport(format!("checkout error: {}", e)))?;

        Ok(PullResult::Success {
            revision: fetch_commit.id().to_string(),
        })
    }

    fn list_remote_links(&self, repo: &str) -> Result<Vec<RemoteLinkInfo>> {
        // Read syncor.toml from the repo
        // For now, clone to temp dir and read
        let temp = tempfile::tempdir()?;
        let cloned = Repository::clone(repo, temp.path())
            .map_err(|e| SyncorError::Transport(format!("clone error: {}", e)))?;

        let toml_path = temp.path().join("syncor.toml");
        if !toml_path.exists() {
            return Ok(Vec::new());
        }

        let content = std::fs::read_to_string(&toml_path)?;
        let doc: toml::Value = toml::from_str(&content)
            .map_err(|e| SyncorError::Config(e.to_string()))?;

        let mut links = Vec::new();
        if let Some(table) = doc.get("links").and_then(|v| v.as_array()) {
            for entry in table {
                if let Some(name) = entry.get("name").and_then(|v| v.as_str()) {
                    links.push(RemoteLinkInfo {
                        name: name.to_string(),
                        created_at: entry
                            .get("created_at")
                            .and_then(|v| v.as_str())
                            .unwrap_or("")
                            .to_string(),
                    });
                }
            }
        }

        Ok(links)
    }

    fn has_remote_changes(&self, link: &LinkInfo) -> Result<bool> {
        let repo = self.open_repo(link)?;

        let mut remote = repo.find_remote("origin")
            .map_err(|e| SyncorError::Transport(format!("find remote error: {}", e)))?;

        // Fetch to update remote refs
        remote.fetch(&["main"], None, None)
            .map_err(|e| SyncorError::Transport(format!("fetch error: {}", e)))?;
        drop(remote);

        let local_head = repo.head().ok().and_then(|h| h.target());
        let remote_head = repo
```

```rust
// crates/syncor-core/src/sync/engine.rs
use crate::config::SyncorPaths;
use crate::daemon::worker::LinkLock;
use crate::error::{Result, SyncorError};
use crate::link::LinkInfo;
use crate::sync::catalog_merge::{checkpoint_wal, merge_catalogs};
use crate::sync::conflict::{detect_conflicts, ManifestMap};
use crate::sync::restore::RestorePipeline;
use crate::sync::save::SavePipeline;
use crate::sync::state::{StateDb, SyncState};
use crate::transport::{PullResult, PushResult, SyncTransport};
use chkpt_core::store::blob::bytes_to_hex;
use chkpt_core::store::catalog::MetadataCatalog;
use std::path::PathBuf;

pub struct SyncEngine {
    paths: SyncorPaths,
    transport: Box<dyn SyncTransport + Send + Sync>,
}

#[derive(Debug)]
pub struct PushSyncResult {
    pub snapshot_id: Option<String>,
    pub pushed: bool,
}

#[derive(Debug)]
pub struct PullSyncResult {
    pub restored: bool,
    pub files_restored: usize,
}

impl SyncEngine {
    pub fn new(paths: SyncorPaths, transport: Box<dyn SyncTransport + Send + Sync>) -> Self {
        Self { paths, transport }
    }

    fn store_dir(&self, link: &LinkInfo) -> PathBuf {
        self.paths
            .link_repo_dir(link.id.as_str())
            .join("stores")
            .join(&link.name)
    }

    fn state_db(&self, link: &LinkInfo) -> Result<StateDb> {
        let path = self.paths.link_state_db(link.id.as_str());
        if let Some(parent) = path.parent() {
            std::fs::create_dir_all(parent)?;
        }
        StateDb::open(path)
    }

    fn manifest_to_map(&self, catalog: &MetadataCatalog, snapshot_id: &str) -> Result<ManifestMap> {
        let manifest = catalog.snapshot_manifest(snapshot_id)?;
        Ok(manifest
            .into_iter()
            .map(|e| (e.path, e.blob_hash))
            .collect())
    }

    pub fn init_link(&self, link: &LinkInfo) -> Result<()> {
        self.transport.init_remote(link)?;
        // Ensure link dir exists
        let link_dir = self.paths.link_dir(link.id.as_str());
        std::fs::create_dir_all(&link_dir)?;
        Ok(())
    }

    pub fn push(&self, link: &LinkInfo) -> Result<PushSyncResult> {
        // Acquire per-link filesystem lock
        let _lock = LinkLock::acquire(&self.paths, link.id.as_str())?;

        let store_dir = self.store_dir(link);

        // Save current state
        let save_result = SavePipeline::run(&link.local_dir, &store_dir, None)?;

        // Checkpoint WAL before pushing
        let catalog_path = store_dir.join("catalog.sqlite");
        if catalog_path.exists() {
            checkpoint_wal(&catalog_path)?;
        }

        // Push
        let push_result = self.transport.push(link, &store_dir)?;

        let db = self.state_db(link)?;
        match push_result {
            PushResult::Success { revision } => {
                let state = SyncState {
                    link_id: link.id.as_str().to_string(),
                    last_local_snapshot: Some(save_result.snapshot_id.clone()),
                    last_remote_revision: Some(revision),
                    last_synced_snapshot_id: Some(save_result.snapshot_id.clone()),
                    last_sync_at: Some(chrono::Utc::now().to_rfc3339()),
                };
                db.upsert_sync_state(&state)?;
                db.append_log(link.id.as_str(), "push", "success", None)?;

                Ok(PushSyncResult {
                    snapshot_id: Some(save_result.snapshot_id),
                    pushed: true,
                })
            }
            PushResult::Conflict { details } => {
                db.append_log(
                    link.id.as_str(),
                    "push",
                    "conflict",
                    Some(&details.message),
                )?;
                Err(SyncorError::Conflict(details.message))
            }
        }
    }

    pub fn pull(&self, link: &LinkInfo) -> Result<PullSyncResult> {
        // Acquire per-link filesystem lock
        let _lock = LinkLock::acquire(&self.paths, link.id.as_str())?;

        let store_dir = self.store_dir(link);
        let catalog_path = store_dir.join("catalog.sqlite");

        // Pull remote
        let pull_result = self.transport.pull(link, &store_dir)?;

        match pull_result {
            PullResult::UpToDate => {
                return Ok(PullSyncResult {
                    restored: false,
                    files_restored: 0,
                });
            }
            PullResult::Conflict { details } => {
                return Err(SyncorError::Conflict(details.message));
            }
            PullResult::Success { revision } => {
                // Merge remote catalog into local catalog.
                let local_catalog_path = self.paths.link_dir(link.id.as_str())
                    .join("catalog.sqlite");
                let remote_catalog_path = catalog_path.clone();

                if !local_catalog_path.exists() {
                    if let Some(parent) = local_catalog_path.parent() {
                        std::fs::create_dir_all(parent)?;
                    }
                    std::fs::copy(&remote_catalog_path, &local_catalog_path)?;
                } else {
                    merge_catalogs(&local_catalog_path, &remote_catalog_path)?;
                }

                let catalog = MetadataCatalog::open(&local_catalog_path)?;
                let latest_remote = catalog.latest_snapshot()?
                    .ok_or_else(|| SyncorError::Other("no snapshots in remote catalog".into()))?;

                let db = self.state_db(link)?;
                let sync_state = db.get_sync_state(link.id.as_str())?;

                // Build three-point comparison maps
                let remote_map: ManifestMap = catalog
                    .snapshot_manifest(&latest_remote.id)?
                    .into_iter()
                    .map(|e| (e.path, e.blob_hash))
                    .collect();

                let base_map: ManifestMap = match sync_state
                    .as_ref()
                    .and_then(|s| s.last_synced_snapshot_id.as_deref())
                {
                    Some(base_id) => catalog
                        .snapshot_manifest(base_id)
                        .unwrap_or_default()
                        .into_iter()
                        .map(|e| (e.path, e.blob_hash))
                        .collect(),
                    None => std::collections::HashMap::new(),
                };

                // Scan local workspace to build local map
                let local_map: ManifestMap = {
                    use chkpt_core::scanner::scan_workspace;
                    use chkpt_core::store::blob::{hash_path_bytes, bytes_to_hex};
                    let scanned = scan_workspace(&link.local_dir, None)?;
                    let mut map = std::collections::HashMap::new();
                    for file in &scanned {
                        let hash = hash_path_bytes(&file.absolute_path, file.is_symlink)?;
                        map.insert(file.relative_path.clone(), hash);
                    }
                    map
                };

                // Three-point conflict detection
                use crate::sync::conflict::{detect_conflicts, FileAction};
                let actions = detect_conflicts(&base_map, &local_map, &remote_map);

                // Check for conflicts
                let conflicts: Vec<_> = actions.iter().filter_map(|a| {
                    if let FileAction::Conflict(c) = a { Some(c.clone()) } else { None }
                }).collect();

                if !conflicts.is_empty() {
                    // Record conflicts and pause
                    for c in &conflicts {
                        db.insert_conflict(&crate::sync::state::ConflictRecord {
                            link_id: link.id.as_str().to_string(),
                            file_path: c.path.clone(),
                            local_hash: c.local_hash.map(|h| bytes_to_hex(&h)),
                            remote_hash: c.remote_hash.map(|h| bytes_to_hex(&h)),
                            base_hash: c.base_hash.map(|h| bytes_to_hex(&h)),
                        })?;
                    }
                    db.append_log(link.id.as_str(), "pull", "conflict",
                        Some(&format!("{} conflicts", conflicts.len())))?;
                    return Err(SyncorError::Conflict(
                        format!("{} file(s) in conflict. Run 'syncor resolve' to fix.", conflicts.len())
                    ));
                }

                // Apply non-conflicting actions
                let pack_set = chkpt_core::store::pack::PackSet::open_all(
                    &store_dir.join("packs"),
                )?;
                let mut files_restored = 0;
                for action in &actions {
                    match action {
                        FileAction::ApplyRemote { path, remote_hash } => {
                            let hash_hex = bytes_to_hex(remote_hash);
                            let content = pack_set.read(&hash_hex)?;
                            let file_path = link.local_dir.join(path);
                            if let Some(parent) = file_path.parent() {
                                std::fs::create_dir_all(parent)?;
                            }
                            std::fs::write(&file_path, content)?;
                            files_restored += 1;
                        }
                        FileAction::DeleteLocal { path } => {
                            let file_path = link.local_dir.join(path);
                            let _ = std::fs::remove_file(&file_path);
                        }
                        FileAction::Conflict(_) => {} // already handled above
                    }
                }

                // Copy merged catalog back to repo dir
                checkpoint_wal(&local_catalog_path)?;
                std::fs::copy(&local_catalog_path, &remote_catalog_path)?;

                let state = SyncState {
                    link_id: link.id.as_str().to_string(),
                    last_local_snapshot: Some(latest_remote.id.clone()),
                    last_remote_revision: Some(revision),
                    last_synced_snapshot_id: Some(latest_remote.id),
                    last_sync_at: Some(chrono::Utc::now().to_rfc3339()),
                };
                db.upsert_sync_state(&state)?;
                db.append_log(link.id.as_str(), "pull", "success", None)?;

                Ok(PullSyncResult {
                    restored: true,
                    files_restored,
                })
            }
        }
    }
}
```

```rust
// crates/syncor-core/src/watch/watcher.rs
use crate::error::{Result, SyncorError};
use notify::{RecommendedWatcher, RecursiveMode, Watcher, EventKind};
use std::path::PathBuf;
use std::time::Duration;
use tokio::sync::mpsc;

#[derive(Debug, Clone)]
pub struct WatchEvent {
    pub changed: bool,
}

pub struct DebouncedWatcher {
    _watcher: RecommendedWatcher,
}

impl DebouncedWatcher {
    pub fn start(
        path: PathBuf,
        debounce: Duration,
        tx: mpsc::Sender<WatchEvent>,
    ) -> Result<Self> {
        let (notify_tx, mut notify_rx) = mpsc::channel::<()>(100);

        let mut watcher = notify::recommended_watcher(move |res: std::result::Result<notify::Event, notify::Error>| {
            if let Ok(event) = res {
                match event.kind {
                    EventKind::Create(_) | EventKind::Modify(_) | EventKind::Remove(_) => {
                        let _ = notify_tx.blocking_send(());
                    }
                    _ => {}
                }
            }
        }).map_err(|e| SyncorError::Other(format!("watcher error: {}", e)))?;

        watcher.watch(&path, RecursiveMode::Recursive)
            .map_err(|e| SyncorError::Other(format!("watch error: {}", e)))?;

        // Debounce task
        tokio::spawn(async move {
            loop {
                // Wait for first event
                if notify_rx.recv().await.is_none() {
                    break;
                }
                // Drain events during debounce window
                let deadline = tokio::time::Instant::now() + debounce;
                loop {
                    match tokio::time::timeout_at(deadline, notify_rx.recv()).await {
                        Ok(Some(())) => continue, // more events, keep draining
                        Ok(None) => return,        // channel closed
                        Err(_) => break,           // debounce window expired
                    }
                }
                // Send debounced event
                if tx.send(WatchEvent { changed: true }).await.is_err() {
                    break;
                }
            }
        });

        Ok(Self { _watcher: watcher })
    }
}
```

```rust
use std::time::Duration;
use tokio::sync::mpsc;
use tokio::task::JoinHandle;

#[derive(Debug, Clone)]
pub struct PollEvent;

pub struct Poller {
    handle: JoinHandle<()>,
    cancel: tokio::sync::watch::Sender<bool>,
}

impl Poller {
    pub fn start(interval: Duration, tx: mpsc::Sender<PollEvent>) -> Self {
        let (cancel_tx, mut cancel_rx) = tokio::sync::watch::channel(false);

        let handle = tokio::spawn(async move {
            let mut interval_timer = tokio::time::interval(interval);
            interval_timer.tick().await; // skip first immediate tick

            loop {
                tokio::select! {
                    _ = interval_timer.tick() => {
                        if tx.send(PollEvent).await.is_err() {
                            break;
                        }
                    }
                    _ = cancel_rx.changed() => {
                        break;
                    }
                }
            }
        });

        Self { handle, cancel: cancel_tx }
    }

    pub fn stop(self) {
        let _ = self.cancel.send(true);
        self.handle.abort();
    }
}
```

```rust
// crates/syncor-core/src/daemon/server.rs
use crate::error::{Result, SyncorError};
use serde::{Deserialize, Serialize};
use serde_json::Value;
use std::path::{Path, PathBuf};
use tokio::io::{AsyncBufReadExt, AsyncWriteExt, BufReader};
use tokio::net::{UnixListener, UnixStream};
use tokio::sync::{mpsc, oneshot};
use tokio::task::JoinHandle;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct IpcRequest {
    pub cmd: String,
    pub args: Value,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct IpcResponse {
    pub ok: bool,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub data: Option<Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub error: Option<String>,
}

impl IpcResponse {
    pub fn ok(data: Value) -> Self {
        Self { ok: true, data: Some(data), error: None }
    }

    pub fn err(msg: impl Into<String>) -> Self {
        Self { ok: false, data: None, error: Some(msg.into()) }
    }
}

pub type CommandSender = mpsc::Sender<(IpcRequest, oneshot::Sender<IpcResponse>)>;

pub struct IpcServer {
    handle: JoinHandle<()>,
    sock_path: PathBuf,
}

impl IpcServer {
    pub async fn start(sock_path: PathBuf, cmd_tx: CommandSender) -> Result<Self> {
        // Remove stale socket
        let _ = std::fs::remove_file(&sock_path);

        let listener = UnixListener::bind(&sock_path)
            .map_err(|e| SyncorError::Other(format!("bind error: {}", e)))?;

        let handle = tokio::spawn(async move {
            loop {
                match listener.accept().await {
                    Ok((stream, _)) => {
                        let tx = cmd_tx.clone();
                        tokio::spawn(handle_connection(stream, tx));
                    }
                    Err(_) => break,
                }
            }
        });

        Ok(Self { handle, sock_path })
    }

    pub async fn stop(self) {
        self.handle.abort();
        let _ = std::fs::remove_file(&self.sock_path);
    }
}

async fn handle_connection(stream: UnixStream, cmd_tx: CommandSender) {
    let (reader, mut writer) = stream.into_split();
    let mut reader = BufReader::new(reader);
    let mut line = String::new();

    while reader.read_line(&mut line).await.unwrap_or(0) > 0 {
        let request: IpcRequest = match serde_json::from_str(line.trim()) {
            Ok(r) => r,
            Err(e) => {
                let resp = IpcResponse::err(format!("invalid request: {}", e));
                let json = serde_json::to_string(&resp).unwrap();
                let _ = writer.write_all(format!("{}\n", json).as_bytes()).await;
                line.clear();
                continue;
            }
        };

        let (reply_tx, reply_rx) = oneshot::channel();
        if cmd_tx.send((request, reply_tx)).await.is_err() {
            break;
        }

        let response = reply_rx.await.unwrap_or_else(|_| {
            IpcResponse::err("handler dropped")
        });

        let json = serde_json::to_string(&response).unwrap();
        if writer.write_all(format!("{}\n", json).as_bytes()).await.is_err() {
            break;
        }
        line.clear();
    }
}

pub struct IpcClient {
    stream: UnixStream,
}

impl IpcClient {
    pub async fn connect(sock_path: &Path) -> Result<Self> {
        let stream = UnixStream::connect(sock_path)
            .await
            .map_err(|_| SyncorError::DaemonNotRunning)?;
        Ok(Self { stream })
    }

    pub async fn send(self, request: IpcRequest) -> Result<IpcResponse> {
        let (reader, mut writer) = self.stream.into_split();

        let json = serde_json::to_string(&request)
            .map_err(|e| SyncorError::Other(e.to_string()))?;
        writer.write_all(format!("{}\n", json).as_bytes()).await?;

        let mut reader = BufReader::new(reader);
        let mut line = String::new();
        reader.read_line(&mut line).await?;

        serde_json::from_str(line.trim())
            .map_err(|e| SyncorError::Other(format!("invalid response: {}", e)))
    }
}
```

```rust
pub struct DaemonManager {
    paths: SyncorPaths,
    config: SyncorConfig,
}

impl DaemonManager {
    pub fn new(paths: SyncorPaths, config: SyncorConfig) -> Self {
        Self { paths, config }
    }

    pub async fn run(&self) -> Result<()> {
        // Write PID file
        let pid = std::process::id();
        std::fs::write(self.paths.pid_file(), pid.to_string())?;

        // Start IPC server
        let (cmd_tx, mut cmd_rx) = mpsc::channel::<(IpcRequest, oneshot::Sender<IpcResponse>)>(32);
        let server = IpcServer::start(self.paths.socket_path(), cmd_tx).await?;

        // Load existing links and start watchers/pollers
        let registry = LinksRegistry::load(&self.paths.links_file()).unwrap_or_default();

        tracing::info!("daemon started with {} links", registry.links.len());

        // Command loop
        while let Some((req, reply_tx)) = cmd_rx.recv().await {
            let response = self.handle_command(&req).await;
            let _ = reply_tx.send(response);
        }

        // Cleanup
        server.stop().await;
        let _ = std::fs::remove_file(self.paths.pid_file());

        Ok(())
    }

    async fn handle_command(&self, req: &IpcRequest) -> IpcResponse {
        match req.cmd.as_str() {
            "status" => {
                let registry = LinksRegistry::load(&self.paths.links_file())
                    .unwrap_or_default();
                let links: Vec<serde_json::Value> = registry.links.iter().map(|l| {
                    json!({
                        "name": l.name,
                        "repo": l.repo,
                        "dir": l.local_dir.to_string_lossy(),
                        "mode": format!("{:?}", l.mode),
                    })
                }).collect();
                IpcResponse::ok(json!({ "links": links }))
            }
            "push" => {
                // TODO: dispatch sync job
                IpcResponse::ok(json!({ "status": "queued" }))
            }
            "pull" => {
                IpcResponse::ok(json!({ "status": "queued" }))
            }
            _ => IpcResponse::err(format!("unknown command: {}", req.cmd)),
        }
    }

    /// Check if daemon is already running.
    pub fn is_running(paths: &SyncorPaths) -> bool {
        let pid_file = paths.pid_file();
        if !pid_file.exists() {
            return false;
        }
        if let Ok(content) = std::fs::read_to_string(&pid_file) {
            if let Ok(pid) = content.trim().parse::<u32>() {
                // Check if process is alive
                unsafe {
                    return libc::kill(pid as i32, 0) == 0;
                }
            }
        }
        false
    }

    /// Clean up stale files from a crashed daemon.
    pub fn cleanup_stale(paths: &SyncorPaths) {
        let _ = std::fs::remove_file(paths.pid_file());
        let _ = std::fs::remove_file(paths.socket_path());
    }
}
```

```rust
async fn cmd_resolve(
    paths: &syncor_core::config::SyncorPaths,
    dir: PathBuf,
) -> Result<()> {
    let dir = std::fs::canonicalize(&dir)?;
    let registry = syncor_core::config::LinksRegistry::load(&paths.links_file())?;
    let link = registry.get_by_dir(&dir)
        .ok_or_else(|| anyhow::anyhow!("directory not linked: {:?}", dir))?;

    let db = syncor_core::sync::state::StateDb::open(
        paths.link_state_db(link.id.as_str()),
    )?;
    let conflicts = db.list_conflicts(link.id.as_str())?;

    if conflicts.is_empty() {
        println!("No conflicts.");
        return Ok(());
    }

    println!("{} conflict(s) found:", conflicts.len());
    for (i, c) in conflicts.iter().enumerate() {
        println!("  [{}] {} — local: {:?}, remote: {:?}",
            i + 1, c.file_path, c.local_hash, c.remote_hash);
    }

    let store_dir = paths.link_repo_dir(link.id.as_str())
        .join("stores")
        .join(&link.name);

    let mut skipped = Vec::new();
    for c in &conflicts {
        let choices = &["(l) keep local", "(r) keep remote", "(s) skip"];
        let selection = dialoguer::Select::new()
            .with_prompt(format!("Resolve '{}'", c.file_path))
            .items(choices)
            .interact()?;
        match selection {
            0 => {
                // Keep local — file already on disk, nothing to do
                println!("  Keeping local for '{}'", c.file_path);
            }
            1 => {
                // Keep remote — restore from pack
                if let Some(ref hash) = c.remote_hash {
                    let packs_dir = store_dir.join("packs");
                    let pack_set = chkpt_core::store::pack::PackSet::open_all(&packs_dir)?;
                    let content = pack_set.read(hash)?;
                    let file_path = link.local_dir.join(&c.file_path);
                    if let Some(parent) = file_path.parent() {
                        std::fs::create_dir_all(parent)?;
                    }
                    std::fs::write(&file_path, content)?;
                    println!("  Restored remote for '{}'", c.file_path);
                } else {
                    // Remote deleted — delete local
                    let file_path = link.local_dir.join(&c.file_path);
                    let _ = std::fs::remove_file(&file_path);
                    println!("  Deleted local for '{}'", c.file_path);
                }
            }
            2 => {
                println!("  Skipping '{}'", c.file_path);
                skipped.push(c.clone());
            }
            _ => {}
        }
    }

    // Only clear resolved conflicts, keep skipped ones
    db.clear_conflicts(link.id.as_str())?;
    for s in &skipped {
        db.insert_conflict(s)?;
    }

    let resolved = conflicts.len() - skipped.len();
    println!("{} resolved, {} skipped.", resolved, skipped.len());
    Ok(())
}
```

```rust
// crates/syncor-core/tests/e2e_test.rs
use syncor_core::config::{SyncorPaths, LinksRegistry};
use syncor_core::link::{LinkId, LinkInfo, LinkMode};
use syncor_core::sync::engine::SyncEngine;
use syncor_core::transport::git::GitTransport;
use tempfile::TempDir;
use std::fs;

/// Full round-trip: Machine A links and pushes, Machine B connects and pulls.
#[test]
fn e2e_push_pull_roundtrip() {
    // Setup
    let remote_dir = TempDir::new().unwrap();
    git2::Repository::init_bare(remote_dir.path()).unwrap();
    let remote_url = remote_dir.path().to_str().unwrap().to_string();

    let workspace_a = TempDir::new().unwrap();
    let workspace_b = TempDir::new().unwrap();
    let data_a = TempDir::new().unwrap();
    let data_b = TempDir::new().unwrap();

    // Machine A: create files and push
    fs::write(workspace_a.path().join("config.yaml"), "key: value").unwrap();
    fs::create_dir(workspace_a.path().join("scripts")).unwrap();
    fs::write(workspace_a.path().join("scripts/deploy.sh"), "#!/bin/bash\necho deploy").unwrap();

    let link_a = LinkInfo {
        id: LinkId::from_parts(&remote_url, "dotfiles"),
        name: "dotfiles".to_string(),
        repo: remote_url.clone(),
        local_dir: workspace_a.path().to_path_buf(),
        mode: LinkMode::Push,
        poll_interval_secs: None,
    };

    let paths_a = SyncorPaths::with_home(data_a.path());
    let transport_a = GitTransport::new(paths_a.clone());
    let engine_a = SyncEngine::new(paths_a, Box::new(transport_a));
    engine_a.init_link(&link_a).unwrap();
    let push_result = engine_a.push(&link_a).unwrap();
    assert!(push_result.pushed);

    // Machine B: pull
    let link_b = LinkInfo {
        id: LinkId::from_parts(&remote_url, "dotfiles"),
        name: "dotfiles".to_string(),
        repo: remote_url.clone(),
        local_dir: workspace_b.path().to_path_buf(),
        mode: LinkMode::Pull,
        poll_interval_secs: None,
    };

    let paths_b = SyncorPaths::with_home(data_b.path());
    let transport_b = GitTransport::new(paths_b.clone());
    let engine_b = SyncEngine::new(paths_b, Box::new(transport_b));
    engine_b.init_link(&link_b).unwrap();
    let pull_result = engine_b.pull(&link_b).unwrap();
    assert!(pull_result.restored);

    // Verify files match
    assert_eq!(
        fs::read_to_string(workspace_b.path().join("config.yaml")).unwrap(),
        "key: value"
    );
    assert_eq!(
        fs::read_to_string(workspace_b.path().join("scripts/deploy.sh")).unwrap(),
        "#!/bin/bash\necho deploy"
    );

    // Machine A: modify and push again
    fs::write(workspace_a.path().join("config.yaml"), "key: updated").unwrap();
    let push2 = engine_a.push(&link_a).unwrap();
    assert!(push2.pushed);

    // Machine B: pull update
    let pull2 = engine_b.pull(&link_b).unwrap();
    assert!(pull2.restored);
    assert_eq!(
        fs::read_to_string(workspace_b.path().join("config.yaml")).unwrap(),
        "key: updated"
    );
}
```
